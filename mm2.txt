-- Helper Functions
local function logMessage(msg, type)
    statusLabel.Text = "Status: " .. msg
end

local function updateButtonUI()
    if isFreecamActive then
        freecamToggleBtn.Text = "Toggle Freecam: ON"
        freecamToggleBtn.BackgroundColor3 = Color3.fromRGB(0, 170, 127)
    else
        freecamToggleBtn.Text = "Toggle Freecam: OFF"
        freecamToggleBtn.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
    end
end

local function getClosestPlayerToMouse()
    local mouse = localPlayer:GetMouse()
    local mousePos = Vector2.new(mouse.X, mouse.Y)
    local closestPlayer = nil
    local shortestDistance = math.huge

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= localPlayer and player.Character then
            local humanoidRootPart = player.Character:FindFirstChild("HumanoidRootPart")
            if humanoidRootPart then
                local screenPos, onScreen = camera:WorldToScreenPoint(humanoidRootPart.Position)
                if onScreen then
                    local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                    if dist < shortestDistance and dist < 100 then
                        shortestDistance = dist
                        closestPlayer = player
                    end
                end
            end
        end
    end

    return closestPlayer
end

local function enableFreecam()
    if isFreecamActive then return end
    isFreecamActive = true

    local character = localPlayer.Character
    if character then
        local hrp = character:FindFirstChild("HumanoidRootPart")
        if hrp then
            currentAngles = Vector2.new(camera.CFrame:ToEulerAnglesYXZ())
        end
    end

    camera.CameraType = Enum.CameraType.Scriptable

    renderConnection = RunService.RenderStepped:Connect(function(dt)
        if not isFreecamActive then return end

        if isSpectating and targetPlayer and targetPlayer.Character then
            local targetHRP = targetPlayer.Character:FindFirstChild("HumanoidRootPart")
            if targetHRP then
                camera.CFrame = targetHRP.CFrame + Vector3.new(0, 5, 10)
                return
            end
        end

        local moveVector = Vector3.new()
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector = moveVector + Vector3.new(0, 0, -1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector = moveVector + Vector3.new(0, 0, 1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector = moveVector + Vector3.new(-1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector = moveVector + Vector3.new(1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.E) then moveVector = moveVector + Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.Q) then moveVector = moveVector + Vector3.new(0, -1, 0) end

        if isRightMouseDown then
            UserInputService.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
            local delta = UserInputService:GetMouseDelta()
            currentAngles = currentAngles - Vector2.new(delta.Y * freecamSens, delta.X * freecamSens)
            currentAngles = Vector2.new(
                math.clamp(currentAngles.X, -math.rad(89), math.rad(89)),
                currentAngles.Y
            )
        else
            UserInputService.MouseBehavior = Enum.MouseBehavior.Default
        end

        local rotCF = CFrame.Angles(0, currentAngles.Y, 0) * CFrame.Angles(currentAngles.X, 0, 0)
        local velocity = rotCF:VectorToWorldSpace(moveVector) * freecamSpeed * dt
        camera.CFrame = CFrame.new(camera.CFrame.Position + velocity) * rotCF
    end)

    updateButtonUI()
    logMessage("Freecam enabled", "Info")
end

local function disableFreecam()
    if not isFreecamActive then return end
    isFreecamActive = false
    isSpectating = false
    isRightMouseDown = false
    targetPlayer = nil

    if renderConnection then
        renderConnection:Disconnect()
        renderConnection = nil
    end

    if ghostCharacter then
        ghostCharacter:Destroy()
        ghostCharacter = nil
    end

    camera.CameraType = Enum.CameraType.Custom
    local character = localPlayer.Character
    if character then
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then
            camera.CameraSubject = humanoid
        end
    end

    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    
    updateButtonUI()
    logMessage("Freecam disabled", "Info")
end

-- Code Match Check Function
local function checkCodeMatch()
    logMessage("Checking code version...", "Info")
    
    task.spawn(function()
        local success, remoteCode = pcall(function()
            return game:HttpGet(SCRIPT_URL)
        end)

        if success and remoteCode then
            -- Read the current script source if running from an environment that supports it, 
            -- or compare against a signature/hash. Since standard LocalScripts can't easily read 
            -- their own source file directly via standard API without getscriptbytecode, 
            -- we can compare text length or contents if bundled, or fetch and compare.
            -- Alternatively, if you want to compare live running behavior or check if remote differs:
            logMessage("Fetched remote code successfully.", "Info")
        else
            logMessage("Failed to fetch remote script.", "Error")
        end
    end)
end

-- UI Interactivity Connections
toggleMenuBtn.MouseButton1Click:Connect(function()
    specFrame.Visible = not specFrame.Visible
end)

freecamToggleBtn.MouseButton1Click:Connect(function()
    if isFreecamActive then
        disableFreecam()
    else
        enableFreecam()
    end
end)

warnBtn.MouseButton1Click:Connect(function()
    if targetPlayer and targetPlayer.Parent then
        logMessage("Warned: " .. targetPlayer.Name .. " (Outdated Version)", "Warning")
        print("[WARNING] Player " .. targetPlayer.Name .. " is using an outdated version!")
    else
        logMessage("No player currently spectated!", "Error")
    end
end)

updateCheckBtn.MouseButton1Click:Connect(function()
    checkCodeMatch()
end)

-- Input Listeners
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not isFreecamActive then return end

    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        isRightMouseDown = true
    elseif input.UserInputType == Enum.UserInputType.MouseButton1 and not gameProcessed then
        local target = getClosestPlayerToMouse()
        if target then
            targetPlayer = target
            isSpectating = true
            logMessage("Spectating: " .. target.Name, "Info")
        else
            isSpectating = false
            targetPlayer = nil
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        isRightMouseDown = false
    end
end)
