local logContainerRef = nil

logMessage = function(text, msgType)
    if not Config.EnableLogConsole then return end
    if not logContainerRef or not logContainerRef.Parent then return end

    local color = Color3.fromRGB(200, 200, 210)
    local prefix = "[INFO] "

    if msgType == "Error" then
        color = Color3.fromRGB(255, 90, 90)
        prefix = "[ERROR] "
    elseif msgType == "Success" then
        color = Color3.fromRGB(46, 204, 113)
        prefix = "[SUCCESS] "
    elseif msgType == "Warn" then
        color = Color3.fromRGB(241, 196, 15)
        prefix = "[WARN] "
    end

    local logItem = Instance.new("TextLabel")
    logItem.Size = UDim2.new(1, 0, 0, 20)
    logItem.BackgroundTransparency = 1
    logItem.Text = prefix .. text
    logItem.TextColor3 = color
    logItem.Font = Enum.Font.Code
    logItem.TextSize = 11
    logItem.TextXAlignment = Enum.TextXAlignment.Left
    logItem.ZIndex = 31
    logItem.Parent = logContainerRef

    local children = logContainerRef:GetChildren()
    local count = 0
    for _, c in ipairs(children) do
        if c:IsA("TextLabel") then count += 1 end
    end
    if count > 50 then
        for _, c in ipairs(children) do
            if c:IsA("TextLabel") then
                c:Destroy()
                break
            end
        end
    end

    logContainerRef.CanvasSize = UDim2.new(0, 0, 0, logContainerRef.UIListLayout.AbsoluteContentSize.Y)
    logContainerRef.CanvasPosition = Vector2.new(0, logContainerRef.CanvasSize.Y)
end

local function applyCyberStyle(parentInstance, cornerRadius, transparency)
    parentInstance.BackgroundColor3 = Color3.fromRGB(15, 15, 20)
    parentInstance.BackgroundTransparency = transparency or 0.22
    parentInstance.BorderSizePixel = 0

    local corner = Instance.new("UICorner")
    corner.CornerRadius = cornerRadius or UDim.new(0, 14)
    corner.Parent = parentInstance

    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(255, 255, 255)
    stroke.Transparency = 0.8
    stroke.Thickness = 1.2
    stroke.Parent = parentInstance

    local gradient = Instance.new("UIGradient")
    gradient.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 255, 255)),
        ColorSequenceKeypoint.new(1, Color3.fromRGB(140, 140, 155))
    })
    gradient.Rotation = 90
    gradient.Parent = parentInstance
    
    return stroke
end

local function applyInteractiveButton(button, defaultColor, activeColor, isActiveCheck)
    button.BackgroundColor3 = defaultColor
    button.BackgroundTransparency = 0.2

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 10)
    corner.Parent = button

    local stroke = Instance.new("UIStroke")
    stroke.Color = Color3.fromRGB(255, 255, 255)
    stroke.Transparency = 0.75
    stroke.Parent = button

    button.MouseEnter:Connect(function()
        local base = isActiveCheck() and activeColor or defaultColor
        local hoverColor = Color3.fromRGB(
            math.clamp(base.R * 255 + 30, 0, 255),
            math.clamp(base.G * 255 + 30, 0, 255),
            math.clamp(base.B * 255 + 30, 0, 255)
        )
        TweenService:Create(button, TweenInfo.new(0.25, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
            BackgroundColor3 = hoverColor,
            BackgroundTransparency = 0.1,
            Size = button.Size + UDim2.new(0, 6, 0, 3)
        }):Play()
    end)
    
    button.MouseLeave:Connect(function()
        local targetColor = isActiveCheck() and activeColor or defaultColor
        TweenService:Create(button, TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            BackgroundColor3 = targetColor,
            BackgroundTransparency = 0.2,
            Size = button.Size - UDim2.new(0, 6, 0, 3)
        }):Play()
    end)

    button.MouseButton1Down:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = button.Size - UDim2.new(0, 4, 0, 2)
        }):Play()
    end)

    button.MouseButton1Up:Connect(function()
        TweenService:Create(button, TweenInfo.new(0.1, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), {
            Size = button.Size + UDim2.new(0, 4, 0, 2)
        }):Play()
    end)
end

local function getPlayerRole(player)
    if not player then return "Innocent", COLOR_INNOCENT end
    
    local hasGun = false
    local hasKnifeRole = false

    local function scanTools(container)
        if not container then return end
        for _, child in ipairs(container:GetChildren()) do
            if child:IsA("Tool") then
                local name = string.lower(child.Name)
                if string.find(name, "gun") or string.find(name, "revolver") then
                    hasGun = true
                elseif string.find(name, "knife") or string.find(name, "kinfe") or string.find(name, "dagger") then
                    hasKnifeRole = true
                end
            end
        end
    end

    scanTools(player.Character)
    scanTools(player:FindFirstChild("Backpack"))

    if hasKnifeRole then
        return "Murder", COLOR_MURDER
    elseif hasGun then
        return "Sheriff", COLOR_SHERIFF
    else
        return "Innocent", COLOR_INNOCENT
    end
end

local function applyHighlight(player, roleText, roleColor)
    local char = player.Character

    end
end

local function getEquippedTool(character)
    if not character then return nil, "None" end
    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("Tool") then
            local name = string.lower(child.Name)
            if string.find(name, "gun") or string.find(name, "revolver") then
                return child, "Gun"
            elseif string.find(name, "knife") or string.find(name, "kinfe") or string.find(name, "dagger") then
                return child, "Knife"
            end
        end
    end
    return nil, "None"
end

local function hasKnife(player)
    local function checkContainer(container)
        if not container then return false end
        for _, child in ipairs(container:GetChildren()) do
            if child:IsA("Tool") then
                local name = string.lower(child.Name)
                if string.find(name, "knife") or string.find(name, "kinfe") or string.find(name, "dagger") then
                    return true
                end
            end
        end
        return false
    end

    if player.Character and checkContainer(player.Character) then return true end
    if player:FindFirstChild("Backpack") and checkContainer(player.Backpack) then return true end
    return false
end

local function hasGun(player)
    local function checkContainer(container)
        if not container then return false end
        for _, child in ipairs(container:GetChildren()) do

                local name = string.lower(child.Name)
                if string.find(name, "gun") or string.find(name, "revolver") then

            end
        end
        return false
    end

    if player.Character and checkContainer(player.Character) then return true end
    if player:FindFirstChild("Backpack") and checkContainer(player.Backpack) then return true end
    return false
end

-- TELEPORT LOOP SEQUENCE (EQUIPS SLOT 1 ON START)
---------------------------------------------------------
startTeleportSequence = function()
    if isTeleportRunning then return end

    local myCharacter = localPlayer.Character
    if not myCharacter or not myCharacter:FindFirstChild("HumanoidRootPart") then return end
    local myHRP = myCharacter.HumanoidRootPart
    local humanoid = myCharacter:FindFirstChildOfClass("Humanoid")

    local backpack = localPlayer:FindFirstChild("Backpack")
    if backpack and humanoid then
        local firstTool = backpack:FindFirstChildOfClass("Tool")
        if firstTool then
            pcall(function()
                humanoid:EquipTool(firstTool)
            end)
            logMessage("Equipped first inventory item for TP loop", "Info")
            task.wait(0.15)
        end
    end

    local activeTool, toolType = getEquippedTool(myCharacter)

    isTeleportRunning = true
    if startButton then
        startButton.Text = "Running..."
        startButton.BackgroundColor3 = Color3.fromRGB(120, 120, 135)
    end

    logMessage("Started Teleport Loop with tool type: " .. toolType, "Info")

    if toolType == "None" then
        logMessage("TP Loop Error: No active tool equipped to execute loop!", "Error")
        myHRP.CFrame = myHRP.CFrame * CFrame.new(0, 100, 0)
        task.wait(5)
    else
        local visitedPlayers = {}

        for _, targetP in ipairs(Players:GetPlayers()) do
            local _, liveType = getEquippedTool(myCharacter)
            if liveType == "None" or not isTeleportRunning then
                logMessage("TP Loop halted or tool unequipped.", "Warn")
                break
            end

            if targetP ~= localPlayer and not visitedPlayers[targetP] then
                visitedPlayers[targetP] = true

                local shouldTarget = (liveType == "Gun" and hasKnife(targetP)) or (liveType == "Knife")

                if shouldTarget then
                    local targetChar = targetP.Character
                    if targetChar and targetChar:FindFirstChild("HumanoidRootPart") then
                        local targetHRP = targetChar.HumanoidRootPart
                        local startTime = os.clock()
                        local lastClick = 0

                        logMessage("Teleporting to target: " .. targetP.Name, "Info")

                        while isTeleportRunning and (os.clock() - startTime < Config.AttackDuration) do
                            local currentTool, liveToolType = getEquippedTool(myCharacter)
                            if liveToolType == "None" then break end
                            if not targetChar or not targetHRP.Parent then break end

                            myHRP.CFrame = targetHRP.CFrame

                            if os.clock() - lastClick >= Config.ClickInterval then
                                if currentTool then
                                    pcall(function()
                                        currentTool:Activate()
                                    end)
                                end
                                lastClick = os.clock()
                            end

                            -- Respect Config.TpDelay between steps
                            if Config.TpDelay > 0 then
                                task.wait(Config.TpDelay)
                            else
                                RunService.Heartbeat:Wait()
                            end
                        end
                    end
                end
            end
        end
    end

    isTeleportRunning = false
    if startButton then
        startButton.Text = "Start TP Loop"
        startButton.BackgroundColor3 = Color3.fromRGB(46, 204, 113)
    end
    logMessage("Teleport Loop finished.", "Info")
end

local function getSpectateTargets()
    local targets = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("Humanoid") then
            table.insert(targets, p)
        end
    end
    return targets

setSpectateTarget = function(index)
    local targets = getSpectateTargets()
    if #targets == 0 then
        isSpectating = false
        targetPlayer = nil
        if nameLabel then nameLabel.Text = "Freecam Mode (No Players)" end
        logMessage("Freecam: No valid spectate targets found.", "Warn")
        return
    end

    if index > #targets then index = 1 end
    if index < 1 then index = #targets end

    spectateIndex = index
    targetPlayer = targets[spectateIndex]
    isSpectating = true
    if nameLabel then nameLabel.Text = "Spectating: " .. targetPlayer.DisplayName end
    logMessage("Now spectating player: " .. targetPlayer.Name, "Info")
end

    if isFreecamActive then return end

    local realCharacter = localPlayer.Character
    if not realCharacter or not realCharacter:FindFirstChild("HumanoidRootPart") then return end

    isFreecamActive = true
    isSpectating = false
    targetPlayer = nil
    if nameLabel then nameLabel.Text = "Free Flying" end

    updateButtonUI()

    ghostCharacter = createGhostModel(realCharacter)

    local rx, ry, _ = camera.CFrame:ToOrientation()
    pitch = rx
    yaw = ry
    cameraPos = camera.CFrame.Position

    camera.CameraType = Enum.CameraType.Scriptable

    renderConnection = RunService.RenderStepped:Connect(function(deltaTime)
        if not isFreecamActive then return end

        if isRightMouseDown then
            local mouseDelta = UserInputService:GetMouseDelta()
            yaw = yaw - math.rad(mouseDelta.X * Config.LookSensitivity)
            pitch = pitch - math.rad(mouseDelta.Y * Config.LookSensitivity)
            pitch = math.clamp(pitch, math.rad(-89), math.rad(89))
        end

        local cameraRotation = CFrame.Angles(0, yaw, 0) * CFrame.Angles(pitch, 0, 0)

        local moveVector = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector += Vector3.new(0, 0, -1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector += Vector3.new(0, 0, 1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector += Vector3.new(-1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector += Vector3.new(1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.E) or UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveVector += Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.Q) or UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveVector += Vector3.new(0, -1, 0) end

        if moveVector.Magnitude > 0 and isSpectating then
            isSpectating = false
            targetPlayer = nil
            if nameLabel then nameLabel.Text = "Free Flying" end
        end

        if isSpectating and targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
            local targetPos = targetPlayer.Character.HumanoidRootPart.Position
            cameraPos = targetPos - (cameraRotation.LookVector * 12) + Vector3.new(0, 3, 0)
        else
            local speed = Config.MoveSpeed
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then
                speed *= 2
            end
            cameraPos += (cameraRotation * moveVector) * speed * deltaTime
        end

        camera.CFrame = CFrame.new(cameraPos) * cameraRotation
    end)
    logMessage("Freecam enabled", "Success")
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        isRightMouseDown = true
        if isFreecamActive then
            UserInputService.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition
        end
    elseif input.KeyCode == Config.ToggleKey and not gameProcessed then
        if isFreecamActive then
            disableFreecam()
        else
            enableFreecam()
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        isRightMouseDown = false
        if isFreecamActive then
            UserInputService.MouseBehavior = Enum.MouseBehavior.Default
        end
    end
end)

localPlayer.CharacterAdded:Connect(function()
    disableFreecam()
    logMessage("Player character respawned, re-initializing UI...", "Info")
    task.wait(1)
    createNexusUI()
end)

createNexusUI()
