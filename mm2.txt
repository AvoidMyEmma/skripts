    label.Parent = row

    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.5, 0, 0.8, 0)
    btn.Position = UDim2.new(0.5, 0, 0.1, 0)
    btn.BackgroundColor3 = Color3.fromRGB(20, 20, 28)
    btn.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
    btn.TextColor3 = Color3.fromRGB(0, 229, 255)
    btn.Text = CONFIG[configKey].Name
    btn.Font = Enum.Font.GothamBold
    btn.TextSize = 11
    btn.ZIndex = 22
    btn.Parent = row

    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = btn

    keybindButtonRefs[configKey] = btn

    btn.MouseButton1Click:Connect(function()
        if listeningForKeyConfig then return end
        listeningForKeyConfig = configKey
        btn.Text = "Press key..."
        btn.TextColor3 = Color3.fromRGB(255, 200, 80)

        local connection
        connection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
            if input.UserInputType == Enum.UserInputType.Keyboard then
                CONFIG[configKey] = input.KeyCode
                btn.Text = input.KeyCode.Name
                btn.TextColor3 = Color3.fromRGB(0, 229, 255)
                listeningForKeyConfig = nil
                updateButtonTexts()
                connection:Disconnect()
            end
        end)
    end)

    row.Parent = configContainer
end

local function processCommand(text)
    text = string.lower(string.match(text, "^%s*(.-)%s*$"))
    if text == "" then return end

    logToConsole("> " .. text, Enum.MessageType.MessageInfo)

    local parts = {}
    for word in string.gmatch(text, "%S+") do
        table.insert(parts, word)
    end

    local cmd = parts[1]

    if cmd == "toggle" then
        local moduleName = parts[2]
        if not moduleName then
            logToConsole("Usage: toggle [antimurder/murderka/sheriffka/freecam/tploop]", Enum.MessageType.MessageWarning)
            return
        end

        if moduleName == "antimurder" or moduleName == "am" then
            isAntiMurderActive = not isAntiMurderActive
            antiMurderBtn.BackgroundColor3 = isAntiMurderActive and Color3.fromRGB(12, 35, 30) or Color3.fromRGB(18, 18, 25)
        elseif moduleName == "murderka" or moduleName == "mka" then
            isMurderKillAuraActive = not isMurderKillAuraActive
            murderKAButton.BackgroundColor3 = isMurderKillAuraActive and Color3.fromRGB(12, 35, 30) or Color3.fromRGB(18, 18, 25)
        elseif moduleName == "sheriffka" or moduleName == "ska" then
            isSheriffKillAuraActive = not isSheriffKillAuraActive
            sheriffKAButton.BackgroundColor3 = isSheriffKillAuraActive and Color3.fromRGB(12, 35, 30) or Color3.fromRGB(18, 18, 25)
        elseif moduleName == "freecam" or moduleName == "fc" then
            if isFreecamActive then disableFreecam() else enableFreecam() end
        elseif moduleName == "tploop" or moduleName == "tp" then
            if not isTeleportRunning then task.spawn(startTeleportSequence) else isTeleportRunning = false end
        else
            logToConsole("Unknown module: " .. moduleName, Enum.MessageType.MessageError)
        end
        updateButtonTexts()

    elseif cmd == "set" then
        local settingName = parts[2]
        local val = tonumber(parts[3])

        if not settingName or not val then
            logToConsole("Usage: set [murderka/sheriffka/antimurder/height/speed/gui_trans/elem_trans] [value]", Enum.MessageType.MessageWarning)
            return
        end

        if settingName == "murderka" or settingName == "mka" then
            CONFIG.MURDER_KA_DISTANCE = val
            if settingTextBoxes["MURDER_KA_DISTANCE"] then settingTextBoxes["MURDER_KA_DISTANCE"].Text = tostring(val) end
        elseif settingName == "sheriffka" or settingName == "ska" then
            CONFIG.SHERIFF_KA_DISTANCE = val
            if settingTextBoxes["SHERIFF_KA_DISTANCE"] then settingTextBoxes["SHERIFF_KA_DISTANCE"].Text = tostring(val) end
        elseif settingName == "antimurder" or settingName == "am" then
            CONFIG.ANTI_MURDER_DISTANCE = val
            if settingTextBoxes["ANTI_MURDER_DISTANCE"] then settingTextBoxes["ANTI_MURDER_DISTANCE"].Text = tostring(val) end
        elseif settingName == "height" or settingName == "sky" then
            CONFIG.ANTI_MURDER_SKY_HEIGHT = val
            if settingTextBoxes["ANTI_MURDER_SKY_HEIGHT"] then settingTextBoxes["ANTI_MURDER_SKY_HEIGHT"].Text = tostring(val) end
        elseif settingName == "speed" then
            CONFIG.MOVE_SPEED = val
            if settingTextBoxes["MOVE_SPEED"] then settingTextBoxes["MOVE_SPEED"].Text = tostring(val) end
        elseif settingName == "gui_trans" then
            CONFIG.GUI_TRANSPARENCY = math.clamp(val, 0, 1)
            mainFrame.BackgroundTransparency = CONFIG.GUI_TRANSPARENCY
            settingsFrame.BackgroundTransparency = CONFIG.GUI_TRANSPARENCY
            if settingTextBoxes["GUI_TRANSPARENCY"] then settingTextBoxes["GUI_TRANSPARENCY"].Text = tostring(CONFIG.GUI_TRANSPARENCY) end
        elseif settingName == "elem_trans" then
            CONFIG.ELEMENT_TRANSPARENCY = math.clamp(val, 0, 1)
            settingsButton.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
            antiMurderBtn.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
            murderKAButton.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
            sheriffKAButton.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
            toggleButton.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
            consoleScroll.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
            commandInput.BackgroundTransparency = CONFIG.ELEMENT_TRANSPARENCY
            if settingTextBoxes["ELEMENT_TRANSPARENCY"] then settingTextBoxes["ELEMENT_TRANSPARENCY"].Text = tostring(CONFIG.ELEMENT_TRANSPARENCY) end
        else
            logToConsole("Unknown setting: " .. settingName, Enum.MessageType.MessageError)
        end
    elseif cmd == "reset" then
        for k, v in pairs(DEFAULT_CONFIG) do
            CONFIG[k] = v
            if settingTextBoxes[k] then
                settingTextBoxes[k].Text = tostring(v)
            end
            if keybindButtonRefs[k] then
                keybindButtonRefs[k].Text = v.Name
            end
        end
        updateButtonTexts()
        mainFrame.BackgroundTransparency = CONFIG.GUI_TRANSPARENCY
        settingsFrame.BackgroundTransparency = CONFIG.GUI_TRANSPARENCY
        logToConsole("All settings reset to defaults via command.", Enum.MessageType.MessageInfo)
    elseif cmd == "help" then
        logToConsole("Commands: toggle [antimurder/murderka/sheriffka/freecam/tploop]", Enum.MessageType.MessageInfo)
        logToConsole("Commands: set [murderka/sheriffka/antimurder/height/speed/gui_trans/elem_trans] [val]", Enum.MessageType.MessageInfo)
        logToConsole("Commands: reset", Enum.MessageType.MessageInfo)
    else
        logToConsole("Unknown command. Type 'help' for info.", Enum.MessageType.MessageError)
    end
end

local function getPlayerRole(player)
    if not player then return "Innocent", COLOR_INNOCENT end
    local hasGun, hasKnifeRole = false, false

    local function scanTools(container)
        if not container then return end
        for _, child in ipairs(container:GetChildren()) do
            if child:IsA("Tool") then
                local name = string.lower(child.Name)
                if name == "gun" then hasGun = true
                elseif name == "knife" or name == "kinfe" then hasKnifeRole = true end
            end
        end
    end

    scanTools(player.Character)
    scanTools(player:FindFirstChild("Backpack"))

    if hasKnifeRole then return "Murder", COLOR_MURDER
    elseif hasGun then return "Sheriff", COLOR_SHERIFF
    else return "Innocent", COLOR_INNOCENT end
end

local function applyHighlight(player, roleText, roleColor)
    local char = player.Character
    if not char then return end

        existingHighlight.FillColor = roleColor
        existingHighlight.OutlineColor = roleColor

        if existingHighlight then existingHighlight:Destroy() end
    end
end

local function isVisibleToPlayer(targetPart)
    local char = localPlayer.Character
    if not char or not char:FindFirstChild("HumanoidRootPart") then return false end
    local myPos = char.HumanoidRootPart.Position
    local targetPos = targetPart.Position
    if (targetPos - myPos).Magnitude > 60 then return false end
    
    local _, onScreen = camera:WorldToViewportPoint(targetPos)
    if not onScreen then return false end
    
    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = Enum.RaycastFilterType.Exclude
    raycastParams.FilterDescendantsInstances = {char, targetPart.Parent}
    raycastParams.IgnoreWater = true
    
    local rayResult = workspace:Raycast(camera.CFrame.Position, targetPos - camera.CFrame.Position, raycastParams)
    return rayResult == nil
end

-- Anti-Murder Loop
task.spawn(function()
    while true do
        task.wait(0.1)
        if isAntiMurderActive then
            local myChar = localPlayer.Character
            if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                local myHRP = myChar.HumanoidRootPart
                local myPos = myHRP.Position
                
                for _, p in ipairs(Players:GetPlayers()) do
                    if p ~= localPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                        local role, _ = getPlayerRole(p)
                        if role == "Murder" then
                            local targetPart = p.Character.HumanoidRootPart
                            if (targetPart.Position - myPos).Magnitude <= CONFIG.ANTI_MURDER_DISTANCE and isVisibleToPlayer(targetPart) then
                                myHRP.CFrame = myHRP.CFrame * CFrame.new(0, CONFIG.ANTI_MURDER_SKY_HEIGHT, 0)
                                task.wait(2)
                                break
                            end
                        end
                    end
                end
            end
        end
    end
end)

                    if targetP ~= localPlayer and targetP.Character and targetP.Character:FindFirstChild("HumanoidRootPart") then
                        local targetHrp = targetP.Character.HumanoidRootPart
                        if (targetHrp.Position - myHrp.Position).Magnitude <= CONFIG.MURDER_KA_DISTANCE then
                            if getEquippedToolType(myChar) ~= "Knife" then
                                local backpack = localPlayer:FindFirstChild("Backpack")
                                if backpack then
                                    for _, tool in ipairs(backpack:GetChildren()) do
                                        if tool:IsA("Tool") and (string.lower(tool.Name) == "knife" or string.lower(tool.Name) == "kinfe") then
                                            tool.Parent = myChar
                                            break
                                        end
                                    end
                                end
                            end
                            local originalCFrame = myHrp.CFrame
                            local attackStart = os.clock()
                            while isMurderKillAuraActive and targetP.Character and targetHrp.Parent and (os.clock() - attackStart < CONFIG.ATTACK_DURATION) do
                                myHrp.CFrame = targetHrp.CFrame * CFrame.new(0, 0, 1.5)
                                simulateLeftClick()
                                RunService.Heartbeat:Wait()
                            end
                            myHrp.CFrame = originalCFrame
                            task.wait(0.2)
                            break
                        end
                    end
                end
            end
        end
    end
end)

-- Sheriff Kill Aura

                            end

                    end
                end
            end
        end
    end

function startTeleportSequence()
    if isTeleportRunning then return end
    local myCharacter = localPlayer.Character or localPlayer.CharacterAdded:Wait()
    local myHRP = myCharacter:WaitForChild("HumanoidRootPart")
    local toolType = getEquippedToolType(myCharacter)

    isTeleportRunning = true
    startButton.Text = "Running... (" .. string.upper(CONFIG.KEY_TP_LOOP.Name) .. ")"
    startButton.BackgroundColor3 = Color3.fromRGB(12, 35, 30)

    if toolType == "None" then
        local currentPos = myHRP.Position
        local raycastParams = RaycastParams.new()
        raycastParams.FilterType = Enum.RaycastFilterType.Exclude
        raycastParams.FilterDescendantsInstances = {myCharacter}
        raycastParams.IgnoreWater = true
        
        local rayResult = workspace:Raycast(Vector3.new(currentPos.X, 5000, currentPos.Z), Vector3.new(0, -10000, 0), raycastParams)
        if rayResult then
            myHRP.CFrame = CFrame.new(rayResult.Position + Vector3.new(0, 3, 0))
        else
            myHRP.CFrame = CFrame.new(currentPos + Vector3.new(0, 5, 0))
        end
        task.wait(5)
    else
        local visited = {}
        for _, targetP in ipairs(Players:GetPlayers()) do
            if getEquippedToolType(myCharacter) == "None" or not isTeleportRunning then break end
            if targetP ~= localPlayer and not visited[targetP] then
                visited[targetP] = true
                if getEquippedToolType(myCharacter) == "Knife" or hasKnife(targetP) then
                    local targetChar = targetP.Character
                    if targetChar and targetChar:FindFirstChild("HumanoidRootPart") then
                        local targetHRP = targetChar.HumanoidRootPart
                        local startTime = os.clock()
                        local lastClick = 0
                        while isTeleportRunning and (os.clock() - startTime < CONFIG.ATTACK_DURATION) do
                            if getEquippedToolType(myCharacter) == "None" or not targetChar or not targetHRP.Parent then break end
                            myHRP.CFrame = targetHRP.CFrame
                            if os.clock() - lastClick >= CONFIG.CLICK_INTERVAL then
                                simulateLeftClick()
                                lastClick = os.clock()
                            end
                            RunService.Heartbeat:Wait()
                        end
                    end
                end
            end
        end
    end

    isTeleportRunning = false
    updateButtonTexts()
    startButton.BackgroundColor3 = Color3.fromRGB(12, 35, 30)
end

local function updateButtonUI()
    if isFreecamActive then
        toggleButton.BackgroundColor3 = Color3.fromRGB(12, 35, 30)
        toggleButton.TextColor3 = Color3.fromRGB(255, 255, 255)
        specFrame.Visible = true
    else
        toggleButton.BackgroundColor3 = Color3.fromRGB(18, 18, 25)
        toggleButton.TextColor3 = Color3.fromRGB(230, 230, 240)
        specFrame.Visible = false
    end
    updateButtonTexts()
end

local function getSpectateTargets()
    local targets = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= localPlayer and p.Character and p.Character:FindFirstChild("Humanoid") then
            table.insert(targets, p)
        end
    end
    return targets
end

local function setSpectateTarget(index)
    local targets = getSpectateTargets()
    if #targets == 0 then
        isSpectating = false
        targetPlayer = nil
        nameLabel.Text = "Freecam Mode (No Players)"
        return
    end
    if index > #targets then index = 1 end
    if index < 1 then index = #targets end
    spectateIndex = index
    targetPlayer = targets[spectateIndex]
    isSpectating = true
    nameLabel.Text = "Spectating: " .. targetPlayer.DisplayName
end

local function createGhostModel(originalChar)
    local hrp = originalChar:FindFirstChild("HumanoidRootPart")
    if not hrp then return nil end
    local ghost = Instance.new("Model")
    ghost.Name = "GhostAvatar"
    local ghostHrp = hrp:Clone()
    ghostHrp.Transparency = 0.2
    ghostHrp.Color = Color3.fromRGB(0, 229, 255)
    ghostHrp.Material = Enum.Material.ForceField
    ghostHrp.CanCollide = false
    ghostHrp.Anchored = true
    ghostHrp.Parent = ghost
    ghost.PrimaryPart = ghostHrp
    ghost.Parent = workspace
    return ghost
end

local function freezeCharacterMovement()
    local character = localPlayer.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if humanoid then humanoid:Move(Vector3.zero, true) end
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if hrp then
        hrp.AssemblyLinearVelocity = Vector3.zero
        hrp.AssemblyAngularVelocity = Vector3.zero
    end
end

local function restoreCameraToPlayer()
    local character = localPlayer.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local hrp = character:FindFirstChild("HumanoidRootPart")
    if humanoid then
        camera.CameraType = Enum.CameraType.Track
        camera.CameraSubject = humanoid
        if hrp then camera.CFrame = hrp.CFrame * CFrame.new(0, 2, 8) end
        task.defer(function()
            camera.CameraType = Enum.CameraType.Custom
            camera.CameraSubject = humanoid
        end)
    end
end

function disableFreecam()
    isFreecamActive = false
    isSpectating = false
    isRightMouseDown = false
    targetPlayer = nil
    if renderConnection then renderConnection:Disconnect(); renderConnection = nil end
    pcall(function() ContextActionService:UnbindAction(ACTION_FREEZE) end)
    if ghostCharacter then ghostCharacter:Destroy(); ghostCharacter = nil end
    restoreCameraToPlayer()
    UserInputService.MouseBehavior = Enum.MouseBehavior.Default
    updateButtonUI()
end

function teleportToFreecam()
    if not isFreecamActive then return end
    local targetCFrame = CFrame.new(cameraPos) * CFrame.Angles(camera.CFrame:ToOrientation())
    local character = localPlayer.Character
    if character and character:FindFirstChild("HumanoidRootPart") then
        character.HumanoidRootPart.CFrame = targetCFrame
    end
    disableFreecam()
end

function enableFreecam()
    if isFreecamActive then return end
    local realCharacter = localPlayer.Character
    if not realCharacter or not realCharacter:FindFirstChild("HumanoidRootPart") then return end

    isFreecamActive = true
    isSpectating = false
    targetPlayer = nil
    nameLabel.Text = "Free Flying"
    updateButtonUI()
    freezeCharacterMovement()

    pcall(function()
        ContextActionService:BindActionAtPriority(
            ACTION_FREEZE, function() return Enum.ContextActionResult.Sink end, false,
            Enum.ContextActionPriority.High.Value,
            Enum.KeyCode.W, Enum.KeyCode.A, Enum.KeyCode.S, Enum.KeyCode.D, Enum.KeyCode.Space, Enum.KeyCode.LeftShift
        )
    end)

    ghostCharacter = createGhostModel(realCharacter)
    local rx, ry, _ = camera.CFrame:ToOrientation()
    pitch, yaw = rx, ry
    cameraPos = camera.CFrame.Position
    camera.CameraType = Enum.CameraType.Scriptable
    UserInputService.MouseBehavior = Enum.MouseBehavior.Default

    renderConnection = RunService.RenderStepped:Connect(function(deltaTime)
        if not isFreecamActive then return end
        freezeCharacterMovement()

        if isRightMouseDown then
            local mouseDelta = UserInputService:GetMouseDelta()
            yaw = yaw - math.rad(mouseDelta.X * CONFIG.LOOK_SENSITIVITY)
            pitch = pitch - math.rad(mouseDelta.Y * CONFIG.LOOK_SENSITIVITY)
            pitch = math.clamp(pitch, math.rad(-89), math.rad(89))
        end

        local cameraRotation = CFrame.Angles(0, yaw, 0) * CFrame.Angles(pitch, 0, 0)
        local moveVector = Vector3.zero
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then moveVector += Vector3.new(0, 0, -1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then moveVector += Vector3.new(0, 0, 1) end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then moveVector += Vector3.new(-1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then moveVector += Vector3.new(1, 0, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then moveVector += Vector3.new(0, 1, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then moveVector += Vector3.new(0, -1, 0) end

        if moveVector.Magnitude > 0 and isSpectating then
            isSpectating = false
            targetPlayer = nil
            nameLabel.Text = "Free Flying"
        end

        if isSpectating and targetPlayer and targetPlayer.Character and targetPlayer.Character:FindFirstChild("HumanoidRootPart") then
            cameraPos = targetPlayer.Character.HumanoidRootPart.Position - (cameraRotation.LookVector * 12) + Vector3.new(0, 3, 0)
        else
            local speed = CONFIG.MOVE_SPEED
            if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then speed *= 2 end
            cameraPos += (cameraRotation * moveVector) * speed * deltaTime
        end

        camera.CFrame = CFrame.new(cameraPos) * cameraRotation
    end)
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if listeningForKeyConfig then return end
    
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        isRightMouseDown = true
        if isFreecamActive then UserInputService.MouseBehavior = Enum.MouseBehavior.LockCurrentPosition end
    elseif input.UserInputType == Enum.UserInputType.MouseButton3 then
        if isFreecamActive then teleportToFreecam() end
    elseif input.UserInputType == Enum.UserInputType.Keyboard and not gameProcessed then
        if input.KeyCode == CONFIG.KEY_TP_LOOP then
            if not isTeleportRunning then task.spawn(startTeleportSequence) else isTeleportRunning = false end
        elseif input.KeyCode == CONFIG.KEY_ANTI_MURDER then
            isAntiMurderActive = not isAntiMurderActive
            antiMurderBtn.BackgroundColor3 = isAntiMurderActive and Color3.fromRGB(12, 35, 30) or Color3.fromRGB(18, 18, 25)
            updateButtonTexts()
        elseif input.KeyCode == CONFIG.KEY_MURDER_KA then
            isMurderKillAuraActive = not isMurderKillAuraActive
            murderKAButton.BackgroundColor3 = isMurderKillAuraActive and Color3.fromRGB(12, 35, 30) or Color3.fromRGB(18, 18, 25)
            updateButtonTexts()
        elseif input.KeyCode == CONFIG.KEY_SHERIFF_KA then
            isSheriffKillAuraActive = not isSheriffKillAuraActive
            sheriffKAButton.BackgroundColor3 = isSheriffKillAuraActive and Color3.fromRGB(12, 35, 30) or Color3.fromRGB(18, 18, 25)
            updateButtonTexts()
        elseif input.KeyCode == CONFIG.KEY_FREECAM then
            if isFreecamActive then disableFreecam() else enableFreecam() end
        end
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton2 then
        isRightMouseDown = false
        if isFreecamActive then UserInputService.MouseBehavior = Enum.MouseBehavior.Default end
    end
end)
