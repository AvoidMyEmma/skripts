---------------------------------------------------------
-- LOGIC & HELPER FUNCTIONS
---------------------------------------------------------

local isRunning = false

-- Check what valid item you currently have EQUIPPED
local function getEquippedToolType(character)
    if not character then return "None" end
    
    for _, child in ipairs(character:GetChildren()) do
        if child:IsA("Tool") then
            local name = string.lower(child.Name)
            if name == "gun" then
                return "Gun"
            elseif name == "knife" or name == "kinfe" then
                return "Knife"
            end
        end
    end
    
    return "None"
end

-- Checks if a target player has a Knife in Character OR Backpack
local function hasKnife(player)
    if player.Character then
        for _, child in ipairs(player.Character:GetChildren()) do
            if child:IsA("Tool") then
                local name = string.lower(child.Name)
                if name == "knife" or name == "kinfe" then
                    return true
                end
            end
        end
    end

    local backpack = player:FindFirstChild("Backpack")
    if backpack then
        for _, child in ipairs(backpack:GetChildren()) do
            if child:IsA("Tool") then
                local name = string.lower(child.Name)
                if name == "knife" or name == "kinfe" then
                    return true
                end
            end
        end
    end

    return false
end

-- Simulate a virtual left mouse click at viewport center
local function simulateLeftClick()
    local viewport = workspace.CurrentCamera.ViewportSize
    local centerX = viewport.X / 2
    local centerY = viewport.Y / 2
    
    VirtualInputManager:SendMouseButtonEvent(centerX, centerY, 0, true, game, 0)
    task.wait(0.01)
    VirtualInputManager:SendMouseButtonEvent(centerX, centerY, 0, false, game, 0)
end

-- Continuous background auto-walk task
task.spawn(function()
    while true do
        task.wait(0.5)
        if isRunning then
            local character = LocalPlayer.Character
            if character then
                local humanoid = character:FindFirstChildOfClass("Humanoid")
                local hrp = character:FindFirstChild("HumanoidRootPart")
                if humanoid and hrp then
                    local randomOffset = Vector3.new(
                        math.random(-30, 30),
                        0,
                        math.random(-30, 30)
                    )
                    humanoid:MoveTo(hrp.Position + randomOffset)
                end
            end
        end
    end
end)

---------------------------------------------------------
-- MAIN TELEPORT SEQUENCE
---------------------------------------------------------

local function startTeleportSequence()
    if isRunning then return end

    local myCharacter = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local myHRP = myCharacter:WaitForChild("HumanoidRootPart")
    local toolType = getEquippedToolType(myCharacter)

    isRunning = true
    startButton.Text = "Running..."
    startButton.BackgroundColor3 = Color3.fromRGB(150, 150, 150)

    if toolType == "None" then
        -- NO ITEM EQUIPPED: TP 100 studs up once
        myHRP.CFrame = myHRP.CFrame * CFrame.new(0, 100, 0)
        task.wait(5)
    else
        -- ITEM EQUIPPED: Single-pass player execution
        local visitedPlayers = {}

        for _, targetPlayer in ipairs(Players:GetPlayers()) do
            local currentTool = getEquippedToolType(myCharacter)
            if currentTool == "None" or not isRunning then
                break
            end

            -- Process each player ONLY ONCE per activation
            if targetPlayer ~= LocalPlayer and not visitedPlayers[targetPlayer] then
                visitedPlayers[targetPlayer] = true

                local shouldTarget = false
                if currentTool == "Gun" then
                    -- Gun Mode: Target ONLY if they have the Knife
                    shouldTarget = hasKnife(targetPlayer)
                elseif currentTool == "Knife" then
                    -- Knife Mode: Target ALL other players
                    shouldTarget = true
                end

                if shouldTarget then
                    local targetChar = targetPlayer.Character
                    if targetChar and targetChar:FindFirstChild("HumanoidRootPart") then
                        local targetHRP = targetChar.HumanoidRootPart
                        local startTime = os.clock()
                        local lastClick = 0

                        -- Lock onto player continuously for ATTACK_DURATION while clicking
                        while isRunning and (os.clock() - startTime < ATTACK_DURATION) do
                            local liveTool = getEquippedToolType(myCharacter)
                            if liveTool == "None" then break end

                            -- Ensure player hasn't died or despawned mid-sequence
                            if not targetChar or not targetHRP.Parent then break end

                            -- Continuously stick to the target's exact position
                            myHRP.CFrame = targetHRP.CFrame

                            -- Click at configured interval
                            if os.clock() - lastClick >= CLICK_INTERVAL then
                                simulateLeftClick()
                                lastClick = os.clock()
                            end

                            RunService.Heartbeat:Wait()
                        end
                    end
                end
            end
        end
    end

    -- Reset UI state
    isRunning = false
    startButton.Text = "Start"
    startButton.BackgroundColor3 = Color3.fromRGB(0, 170, 127)
end

-- Connect Buttons
startButton.MouseButton1Click:Connect(startTeleportSequence)

closeButton.MouseButton1Click:Connect(function()
    isRunning = false
    screenGui:Destroy()
end)
